from typing import Optional

def normalize(value: float, scale: float) -> Optional[float]:
    return value / scale if scale != 0 else 0.0

def format_val(val: float) -> str:
    return f"{val:.1f}"

def run_norm() -> None:

    v2 = normalize(20.0, 4.0)
